(function () {
  var stage = document.getElementById('stage');
  var posters = { en: document.getElementById('poster-en'), es: document.getElementById('poster-es') };
  var label = document.getElementById('lang-label');
  var current = 'en';
  function fit() {
    var p = posters[current];
    var s = Math.min(1, stage.clientWidth / 1200);
    p.style.transform = 'scale(' + s + ')';
    stage.style.height = Math.ceil(p.offsetHeight * s) + 'px';
  }
  function setLang(lang) {
    current = lang === 'es' ? 'es' : 'en';
    posters.en.hidden = current !== 'en';
    posters.es.hidden = current !== 'es';
    document.documentElement.lang = current;
    document.title = current === 'es' ? 'Search funds en Colombia | Arista Capital Partners' : 'Search funds in Colombia | Arista Capital Partners';
    label.textContent = current === 'es' ? 'Idioma' : 'Language'; document.getElementById('home-link').textContent = current === 'es' ? 'Volver al inicio' : 'Back to home';
    document.querySelectorAll('.seg button').forEach(function (b) { b.setAttribute('aria-pressed', String(b.dataset.lang === current)); });
    fit();
  }
  function fromHash() { var h = location.hash.slice(1); return h === 'es' || h === 'en' ? h : null; }
  function stored() { try { return localStorage.getItem('acp-lang') || localStorage.getItem('lang'); } catch (e) { return null; } }
  function fromBrowser() {
    var langs = navigator.languages && navigator.languages.length ? navigator.languages : [navigator.language || 'en'];
    return String(langs[0] || 'en').toLowerCase().indexOf('es') === 0 ? 'es' : 'en';
  }
  document.querySelectorAll('.seg button').forEach(function (b) {
    b.addEventListener('click', function () {
      try { localStorage.setItem('acp-lang', b.dataset.lang); } catch (e) {}
      if (location.hash !== '#' + b.dataset.lang) location.hash = b.dataset.lang; else setLang(b.dataset.lang);
    });
  });
  window.addEventListener('hashchange', function () { var h = fromHash(); if (h) setLang(h); });
  window.addEventListener('resize', fit);
  if (document.fonts && document.fonts.ready) document.fonts.ready.then(fit);
  setLang(fromHash() || stored() || fromBrowser());
})();
